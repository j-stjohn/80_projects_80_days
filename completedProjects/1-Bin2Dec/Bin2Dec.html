Bin2Dec

Convert Binary Number into human-readable integer


Here is the code in Python

I originally did this in Python as it was a language I am already familar with, but I found this as an opportunity to learn Javascript.

Here is the code in JavaScript

  1. // Create a function to check if the input is a binary number
  2. checkIfBin(binaryValue) {
  3. // Run a for loop through each character in the string
  4. for (let i = 0; i < binaryValue.length; i = i + 1) {
  5. // Return 1 on success, 0 on fail
  6. if (Number(binaryValue[i]) <= 1 & Number(binaryValue[i]) >= 0) {
  7. return 1;
  8. }
  9. else {
  10. return 0;
  11. }
  12. }
  13. }
  15. // Function to convert binary number into integer
  16. function bin2dec(binaryValue) {
  17. // Run binary number check
  18. if (checkIfBin(binaryValue) == 1) {
  19. // Establish variables into memory
  20. let i = 0;
  21. let output = 0;
  22. // while loop that only iterates the amount of times for the char length of string
  23. while(i < binaryValue.length) {
  24. // Binary conversion function
  25. output = output + (Number(binaryValue[i]) * (2 ** (binaryValue.length - i - 1)));
  26. i = i + 1;
  27. }
  28. return "Output: " + output;
  29. }
  30. // Error Message
  31. else {
  32. return "ERROR: Please Input a binary number."
  33. }
  34. }
  36. // Create input function from form
  37. function takeInput() {
  38. // Establish variables into memory
  39. const form = document.getElementById("testForm");
  40. const binaryInput = document.getElementById("binInput");
  41. const outputLocation = document.getElementById("testingOutput");
  42. const outputText = document.getElementById("testingOutputText");
  43. // Create a listener event
  44. form.addEventListener('submit', function(event) {
  45. // Stop the page from refreshing during function
  46. event.preventDefault();
  47. // Create variable of input
  48. const binaryValue = binaryInput.value;
  49. // Set output to equal the input into our function
  50. outputText.textContent = bin2dec(binaryValue);
  51. // Display styling
  52. outputLocation.style.display = 'block';
  53. // Turn out input field blank
  54. binaryInput.value = '';
  55. });
  56. }
  58. // Create main function
  59. function main() {
  60. takeInput();
  61. }
  63. // Run Program
  64. main()
